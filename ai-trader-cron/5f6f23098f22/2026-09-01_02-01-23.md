# Daily Hermes backup to GitHub

REPO_DIR="/tmp/hermes-backup"
SOURCE_DIR="/home/vivian/.hermes"
GIT_TOKEN=$(cat $HOME/.config/hermes/backup_pat)

# Ensure repo exists

    git clone "https://${GIT_TOKEN}@github.com/vivianyeah/hermes.git" "$REPO_DIR"
fi

cd "$REPO_DIR"
git pull origin main

# Sync files (exclude cache/logs)
rsync -av --exclude='.git' --exclude='cache/' --exclude='audio_cache/' --exclude='*.log' --exclude='gateway.pid' --exclude='gateway.lock' --exclude='state.db' --exclude='state.db-shm' --exclude='state.db-wal' "$SOURCE_DIR/" "$REPO_DIR/"

# Commit and push

git commit -m "Auto backup $(date '+%Y-%m-%d %H:%M')"
git push origin main
