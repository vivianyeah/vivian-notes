## Hardened replacement

```bash
#!/bin/bash
# Daily Hermes backup to GitHub — hardened
set -euo pipefail

TOKEN_FILE="${HOME}/.config/hermes-backup/token"
[ -f "$TOKEN_FILE" ] || { echo "ERROR: missing $TOKEN_FILE" >&2; exit 1; }
GH_TOKEN="$(cat "$TOKEN_FILE")"

REPO_DIR="/tmp/hermes-backup"
SOURCE_DIR="${HOME}/.hermes"
REPO_URL="https://x-access-token:${GH_TOKEN}@github.com/vivianyeah/hermes.git"
LOG_FILE="${HOME}/.local/share/hermes-backup/backup.log"
LOCK_FILE="/tmp/hermes-backup.lock"

mkdir -p "$(dirname "$LOG_FILE")"
exec >> "$LOG_FILE" 2>&1
echo "=== Backup started: $(date) ==="

# Prevent overlapping runs
exec 9>"$LOCK_FILE"
flock -n 9 || { echo "Another backup is running, exiting"; exit 0; }

# Ensure repo exists
if [ ! -d "$REPO_DIR/.git" ]; then
    git clone "$REPO_URL" "$REPO_DIR"
fi
cd "$REPO_DIR"

# Ensure git identity exists for this repo
git config user.name  "Hermes Backup"      2>/dev/null || true
git config user.email "backup@hermes.local" 2>/dev/null || true

# Pull latest
timeout 60 git pull origin main || echo "WARN: git pull failed, continuing"

# Sync files
rsync -av --delete \
    --exclude='.git' \
    --exclude='cache/' \
    --exclude='audio_cache/' \
    --exclude='*.log' \
    --exclude='gateway.pid' \
    --exclude='gateway.lock' \
    --exclude='state.db' \
    --exclude='state.db-shm' \
    --exclude='state.db-wal' \
    --exclude='credentials*' \
    --exclude='*.pem' \
    --exclude='*.key' \
    "$SOURCE_DIR/" "$REPO_DIR/"

# Commit only if there are changes
if [ -n "$(git status --porcelain)" ]; then
    git add -A
    git commit -m "Auto backup $(date '+%Y-%m-%d %H:%M')"
    timeout 60 git push origin main || echo "ERROR: git push failed"
else
    echo "No changes to commit"
fi

echo "=== Backup completed: $(date) ==="
```
